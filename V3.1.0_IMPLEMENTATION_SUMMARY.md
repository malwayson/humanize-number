## Examples Usage

### Currency Symbols

```typescript
humanizeCurrency(1500000, { currencySymbol: "€" }); // "€1.50 M"
humanizeCurrency(2500000, {
  currencySymbol: "EUR",
  currencyPosition: "suffix",
}); // "2.50 M EUR"
```

### Plugin System

```typescript
registerPlugin(examplePlugins.seismic);
humanizeNumber(7.5, "seismic"); // "7.5 M"

registerPlugin(examplePlugins.radiation);
humanizeNumber(0.001, "radiation"); // "1.00 mSv"
```

### Custom Plugin

```typescript
registerPlugin({
  name: "windSpeed",
  formatMethod: "wind-speed",
  units: {
    metric: [
      { value: 1000 / 3600, symbol: "km/h", name: "kilometers per hour" },
    ],
    imperial: [{ value: 0.44704, symbol: "mph", name: "miles per hour" }],
  },
  defaultOptions: { precision: 1 },
});

humanizeNumber(15, "wind-speed"); // "34 mph"
```
