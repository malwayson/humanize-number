## Plugin System

v3.1.0 introduces a powerful plugin system that allows you to register custom format methods.

### Basic Plugin Registration

```typescript
import { registerPlugin, humanizeNumber } from "@malwayson/humanize-number";

// Define a custom plugin
const seismicPlugin = {
  name: "seismic",
  formatMethod: "seismic",
  units: {
    metric: [{ value: 1, symbol: "M", name: "magnitude" }],
    imperial: [{ value: 1, symbol: "M", name: "magnitude" }],
  },
  defaultOptions: {
    precision: 1,
  },
};

// Register the plugin
registerPlugin(seismicPlugin);

// Use it
console.log(humanizeNumber(7.5, "seismic")); // "7.5 M"
console.log(humanizeNumber(6.8, "seismic")); // "6.8 M"
```

### Managing Plugins

```typescript
import {
  registerPlugin,
  unregisterPlugin,
  getPlugin,
  hasPlugin,
  getAllPlugins,
  clearPlugins,
} from "@malwayson/humanize-number";

// Check if plugin exists
if (hasPlugin("seismic")) {
  console.log("Seismic plugin is registered");
}

// Get a specific plugin
const plugin = getPlugin("seismic");

// Get all registered plugins
const allPlugins = getAllPlugins();
console.log(`${allPlugins.length} plugins registered`);

// Unregister a plugin
unregisterPlugin("seismic");

// Clear all plugins
clearPlugins();
```

## Currency Symbol Support

v3.1.0 adds customizable currency symbols and positioning.

### Basic Usage

```typescript
import { humanizeCurrency } from "@malwayson/humanize-number";

// Default (USD)
console.log(humanizeCurrency(1500)); // "$1.50 K"
console.log(humanizeCurrency(2500000)); // "$2.50 M"
console.log(humanizeCurrency(1000000000)); // "$1.00 B"
```

### Custom Currency Symbols

```typescript
// Euro
console.log(humanizeCurrency(1500, { currencySymbol: "€" })); // "€1.50 K"

// British Pound
console.log(humanizeCurrency(2500000, { currencySymbol: "£" })); // "£2.50 M"

// Japanese Yen
console.log(humanizeCurrency(1000000, { currencySymbol: "¥" })); // "¥1.00 M"

// Indian Rupee
console.log(humanizeCurrency(500000, { currencySymbol: "₹" })); // "₹500.00 K"

// Bitcoin
console.log(humanizeCurrency(50000, { currencySymbol: "₿" })); // "₿50.00 K"

// Custom text
console.log(humanizeCurrency(1500, { currencySymbol: "USD" })); // "USD1.50 K"
```

### Currency Position

```typescript
// Prefix (default)
console.log(humanizeCurrency(1500, { currencySymbol: "$" })); // "$1.50 K"

// Suffix
console.log(
  humanizeCurrency(1500, {
    currencySymbol: "€",
    currencyPosition: "suffix",
  })
); // "1.50 K €"

console.log(
  humanizeCurrency(2500000, {
    currencySymbol: "CHF",
    currencyPosition: "suffix",
  })
); // "2.50 M CHF"
```

### Combined Options

```typescript
// High precision with Euro
console.log(
  humanizeCurrency(1234567, {
    currencySymbol: "€",
    precision: 3,
  })
); // "€1.235 M"

// Approximate values
console.log(
  humanizeCurrency(1500000, {
    currencySymbol: "$",
    approximate: true,
  })
); // "~$1.50 M"

// Negative numbers
console.log(
  humanizeCurrency(-5000, {
    currencySymbol: "£",
  })
); // "-£5.00 K"

// Show sign for positive
console.log(
  humanizeCurrency(3000, {
    currencySymbol: "$",
    showSign: true,
  })
); // "+$3.00 K"

// Verbose units
console.log(
  humanizeCurrency(2500000, {
    currencySymbol: "$",
    verboseUnits: true,
  })
); // "$2.50 million"

// European formatting
console.log(
  humanizeCurrency(1234567, {
    currencySymbol: "€",
    delimiter: ".",
    separator: ",",
  })
); // "€1,23 M"
```

## Creating Custom Plugins

### Simple Plugin Example

```typescript
import { registerPlugin, FormatPlugin } from "@malwayson/humanize-number";

const windSpeedPlugin: FormatPlugin = {
  name: "windSpeed",
  formatMethod: "wind-speed",
  units: {
    metric: [
      { value: 1000 / 3600, symbol: "km/h", name: "kilometers per hour" },
      { value: 1, symbol: "m/s", name: "meters per second" },
    ],
    imperial: [
      { value: 0.44704, symbol: "mph", name: "miles per hour" },
      { value: 0.514444, symbol: "kn", name: "knots" },
    ],
  },
  defaultOptions: {
    precision: 1,
    unitSystem: "imperial",
  },
};

registerPlugin(windSpeedPlugin);

console.log(humanizeNumber(15, "wind-speed")); // "34 mph"
console.log(humanizeNumber(25, "wind-speed", { unitSystem: "metric" })); // "90 km/h"
```

### Plugin with Custom Formatter

```typescript
const beaufortPlugin: FormatPlugin = {
  name: "beaufort",
  formatMethod: "beaufort",
  units: {
    metric: [{ value: 1, symbol: "Bf", name: "Beaufort" }],
    imperial: [{ value: 1, symbol: "Bf", name: "Beaufort" }],
  },
  defaultOptions: {
    precision: 0,
  },
  formatter: (value, unit, options) => {
    // Beaufort scale: 0-12
    const beaufort = Math.round(Math.pow(value / 0.836, 2 / 3));
    const clamped = Math.max(0, Math.min(12, beaufort));

    const descriptions = [
      "Calm",
      "Light air",
      "Light breeze",
      "Gentle breeze",
      "Moderate breeze",
      "Fresh breeze",
      "Strong breeze",
      "High wind",
      "Gale",
      "Strong gale",
      "Storm",
      "Violent storm",
      "Hurricane",
    ];

    return `${clamped} ${unit.symbol} (${descriptions[clamped]})`;
  },
};

registerPlugin(beaufortPlugin);

console.log(humanizeNumber(5, "beaufort")); // "3 Bf (Gentle breeze)"
console.log(humanizeNumber(20, "beaufort")); // "8 Bf (Gale)"
```

### Plugin with Parser

```typescript
const customPlugin: FormatPlugin = {
  name: "custom",
  formatMethod: "custom",
  units: {
    metric: [
      { value: 1000, symbol: "K", name: "thousand" },
      { value: 1, symbol: "", name: "" },
    ],
    imperial: [
      { value: 1000, symbol: "K", name: "thousand" },
      { value: 1, symbol: "", name: "" },
    ],
  },
  parser: (str: string) => {
    const match = str.match(/^([\d.]+)\s*([KMB])?$/i);
    if (!match) return null;

    const num = parseFloat(match[1]);
    const suffix = match[2]?.toUpperCase();

    const multipliers: Record<string, number> = {
      K: 1000,
      M: 1000000,
      B: 1000000000,
    };

    return suffix ? num * multipliers[suffix] : num;
  },
};

registerPlugin(customPlugin);

// Use the parser
import { parseWithPlugin } from "@malwayson/humanize-number";

console.log(parseWithPlugin("1.5 K", "custom")); // 1500
console.log(parseWithPlugin("2.5 M", "custom")); // 2500000
```

## Plugin Best Practices

### 1. Clear Naming

```typescript
// Good
const radiationPlugin = {
  name: "radiation",
  formatMethod: "radiation",
  // ...
};

// Bad
const myPlugin = {
  name: "p1",
  formatMethod: "fmt",
  // ...
};
```

### 2. Provide Default Options

```typescript
const goodPlugin: FormatPlugin = {
  name: "example",
  formatMethod: "example",
  units: {
    /* ... */
  },
  defaultOptions: {
    precision: 2,
    unitSystem: "metric",
  },
};
```

### 3. Include Both Unit Systems

```typescript
const completePlugin: FormatPlugin = {
  name: "complete",
  formatMethod: "complete",
  units: {
    metric: [
      /* metric units */
    ],
    imperial: [
      /* imperial units */
    ],
  },
};
```

### 4. Add Custom Formatter for Complex Logic

```typescript
const complexPlugin: FormatPlugin = {
  name: "complex",
  formatMethod: "complex",
  units: {
    /* ... */
  },
  formatter: (value, unit, options) => {
    // Complex formatting logic
    if (value < 0) return "Invalid";
    if (value > 1000) return "Very large";
    return `${value.toFixed(1)} ${unit.symbol}`;
  },
};
```

### 5. Provide Parser for Bidirectional Conversion

```typescript
const bidirectionalPlugin: FormatPlugin = {
  name: "bidirectional",
  formatMethod: "bidirectional",
  units: {
    /* ... */
  },
  parser: (str: string) => {
    // Parse logic
    const match = str.match(/regex/);
    return match ? parseFloat(match[1]) : null;
  },
};
```

## Migration from v3.0 to v3.1

### No Breaking Changes

All v3.0 code continues to work without modifications.

### New Features are Opt-in

```typescript
// v3.0 code still works
humanizeCurrency(1500); // "$1.50 K"

// v3.1 features are optional
humanizeCurrency(1500, { currencySymbol: "€" }); // "€1.50 K"
```

### Plugins Don't Affect Core Functionality

```typescript
// Core formats work independently of plugins
humanizeData(1024); // "1.00 KB"

// Plugins add new capabilities
registerPlugin(seismicPlugin);
humanizeNumber(7.5, "seismic"); // "7.5 M"
```
